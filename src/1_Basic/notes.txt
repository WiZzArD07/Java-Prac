Increment and Decrement

        x = 10;
        System.out.println(x++);
        System.out.println(x);

        //output : 10  11

        x = 10;
        System.out.println(++x);
        System.out.println(x);

        //output : 11  11

        int z = 10;
        int y = z++;
        System.out.println(z+" "+y);

        //output : 11  10

        int z = 10;
        int y = ++z;
        System.out.println(z+" "+y);

        //output : 11  11
